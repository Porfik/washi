# washi: настройки (/mnt/ext1/washi/config.txt)
# Большинство ключей можно менять на книге: вкладка «Настройки» (там же объяснено, что они делают).
# Параметры книги применяются при её следующем открытии.
# Где искать книги (вкладка «Библиотека»); в настройках на экране не меняется
library_root=/mnt/ext1
# Настройки → Чтение → Полная перерисовка: полное обновление экрана раз в N перелистываний (0 — никогда)
full_refresh=6
# Настройки → Скорость и память → Кэш страниц: готовые страницы, МБ (страница 1404×1872 ≈ 2,5 МБ)
cache_mb=64
# Настройки → Скорость и память → Готовить вперёд / назад: сколько страниц рисовать заранее
prefetch_ahead=2
prefetch_behind=1
# Настройки → Скорость и память → Потоки отрисовки: по копии документа на поток; у B288 два ядра
render_threads=2
# Настройки → Скорость и память → Соседние страницы пока ждёте.
# 0 — пока нужная страница не готова, соседние не начинать (ей все ядра); 1 — рисовать и их
prefetch_while_waiting=0
# Настройки → Скорость и память → Пропуск страниц при быстром листании.
# Сколько мс экран занят после обновления: нажатия за это время не показываются по одной
# странице, а сдвигают цель, и показывается та, на которой остановились (0 — каждую)
screen_busy_ms=470
# Настройки → Скорость и память → Кэш картинок и шрифтов: кэш ресурсов MuPDF, МБ (на каждый поток)
store_mb=64
# Настройки → Картинки в книгах → Упрощённый цвет: 1 — без ICC-профилей;
# на 16 оттенках разницы не видно, а быстрее на 20 %
no_icc=1
# JPX: распаковка сразу в уменьшенном размере и в 2 потока (патч MuPDF 0002); только здесь
jpx_reduce=1
jpx_threads=2
# Настройки → Картинки в книгах → Фото и сканы: распаковка.
# JPX (фон и фото в книгах из DjVu) можно распаковать мельче нужного — до этого процента.
# 100 — всегда не меньше нужного (качество как у MuPDF); 70 — в 2–3 раза быстрее, чуть мягче фото
jpx_min_scale=70
# 1 — картинки, увеличиваемые до 2 раз, масштабировать заранее, а не попиксельно при наложении; только здесь
upscale_scaler=1
# Настройки → Картинки в книгах → Цветные JPEG сразу в серый: по яркости из самого JPEG,
# без перевода через RGB (фото-книги в 1,5–1,7 раза быстрее; отличие от MuPDF — несколько
# уровней из 256). 0 — как MuPDF. Работает только при no_icc=1
jpeg_gray=1
# Настройки → Чтение → Размер шрифта (EPUB, FB2, HTML), пт; у PDF не используется
reflow_em=28
# Подсказка по жестам (Настройки → Чтение): ключ hint_shown появляется сам —
# 0 — показать при открытии книги, 1 — уже показана
# Сглаживание MuPDF, бит 0..8 (-1 — по умолчанию); только здесь, обычно не трогать
# aa=-1
# Хвосты имён файлов, которые не показывать в названиях книг, через «|», без учёта регистра:
# «Книга (scan).pdf» с title_strip=(scan)|(ocr) — «Книга». Только здесь.
# title_strip=
# Общая папка загрузок: куда класть книги, принятые по Wi-Fi и скачанные из каталогов.
# Меняется в Настройках → Устройство → Папка для загрузок (тогда пишется сюда само).
# Путь от library_root («Downloads», «Книги/Новые», «.» — сама библиотека) или полный путь.
# По умолчанию — Downloads. У приёма по Wi-Fi и у каталогов можно выбрать свою папку вместо общей.
# download_dir=Downloads
